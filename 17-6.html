Promise + async 包饺子

用 setTimeout 模拟每个环节的耗时，串行用 await，并行用 Promise.all

流程：买原料 → 取原料 →（和面 ∥ 拌馅）→ 包饺子 → 吃饺子

核心代码

const wait = sec => new Promise(res => setTimeout(res, sec * 1000));

async function makeDumplings() {
  await wait(6);   // ① 买原料
  await wait(2);   // ② 取原料
  await Promise.all([ // ③ 和面与拌馅同时进行
    wait(5), wait(4)
  ]);
  await wait(3);   // ④ 包饺子
  await wait(2);   // ⑤ 吃饺子
}
